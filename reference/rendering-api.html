SceneRenderer

class SceneRenderer {
  // State
  boundSceneId: string | null
  boundSlideId: string | null
  boundScene: Scene | null
  boundCamera: SceneNode | null
  renderedNodeCount: number

  // Lifecycle
  bind(scene, slideId): void

  // Event handlers
  handleNodeCreated(nodeId): void
  handleNodeRemoved(nodeId): void
  handleNodeReparented(nodeId): void
  handleNodeOrderChanged(nodeId): void
  handleTransformChanged(nodeId): void
  handleKeyframeChanged(nodeId): void
  handleVisibilityChanged(nodeId): void
  handleOpacityChanged(nodeId): void
  handleNodeRenamed(nodeId): void
  handleMaterialChanged(nodeId): void
  handleMeshChanged(nodeId): void
  handleTableChanged(nodeId): void
  handleChartChanged(nodeId): void

  // Time-dependent
  handleTimeChanged(): void        // re-evaluate ALL nodes
  applyIKOverrides(rotations): void
  applyConstraintOverrides(): void
  refreshDeformedMeshSizes(): void

  // Queries
  nodeSize(nodeId): WorldSize | null
  previewTransform(nodeId, x, y): void
}

Evaluation Pipeline

#evaluateAndApply(nodeId): 1. engine.evaluateNode(nodeId, time) → transform + opacity 2. engine.evaluateMaterialOverrides() → tint + uniforms 3. resolveMaterial() → effective tint 4. resolveShader() → shader source + uniforms 5. Compare with lastEvaluated/lastMaterials 6. If changed: applyEvaluatedState(container, state) 7. If material changed: applyMaterialTint(container, tint) 8. Store for next comparison
