IK & Constraints API

Quick lookup for IK chains, solvers, and constraint types.

IKChain

class IKChain {
  id: string
  slideId: string
  boneIds: readonly string[]    // ancestor → descendant
  target: BoneIKTarget
  poleTarget: PoleTarget | null
  ghostNodeId: string | null

  rootBoneId: string             // boneIds[0]
  endBoneId: string              // boneIds[last]
  chainLength: number

  validate(nodeLookup): string | null
}

interface BoneIKTarget {
  position: { x: number, y: number }
  nodeId?: string               // attach to node's world position
}

interface PoleTarget {
  position: { x: number, y: number }
}

IK Solvers

// Analytical (2 bones only)
solveTwoBoneIK(boneNodes, target, poleTarget, getLocalTransform, boneLengths)
  → { rotations: [bone1Local, bone2Local] }

// Iterative (N bones, CCD)
solveCCDIK(boneNodes, target, poleTarget, getLocalTransform, boneLengths, iterations?, tolerance?)
  → { rotations: [bone1Local, ..., boneNLocal] }

// Defaults: iterations=10, tolerance=0.001
// Max rotation per iteration: 0.5 rad (clamped)

Constraint Types

type ConstraintType = 'rotationLimit' | 'lookAt' | 'distance' | 'parent'

interface Constraint {
  id: string
  type: ConstraintType
  priority: number              // lower = applied first
  params: ConstraintParams
}

Rotation Limit

// Requires bone node
RotationLimitParams {
  minRotation: number   // degrees
  maxRotation: number   // degrees
}

// Effect: clamp world rotation to [min, max] (converted to radians)

Look At

// Requires bone node
LookAtParams {
  targetX: number
  targetY: number
  targetNodeId?: string   // use node's world position if provided
}

// Effect: rotation = atan2(targetY - world.y, targetX - world.x)

Distance

// Works on any node
DistanceParams {
  targetNodeId: string
  minDistance: number
  maxDistance: number
}

// Effect: clamp distance from target to [min, max]
//   if too close: move outward along angle
//   if too far: move inward along angle

Parent

// Works on any node
ParentConstraintParams {
  targetNodeId: string
  positionInfluence: number   // 0..1
  rotationInfluence: number   // 0..1
  scaleInfluence: number      // 0..1
}

// Effect: blend toward target
//   x = world.x + (target.x - world.x) * influence
//   0 = no effect, 1 = fully follows

ConstraintManager

class ConstraintManager {
  addConstraint(nodeId, type, priority, params): Constraint
  removeConstraint(nodeId, constraintId): Constraint
  setConstraintParams(nodeId, constraintId, params): void
  getConstraint(constraintId): Constraint
  getConstraintsForNode(nodeId): Constraint[]  // sorted by priority
  removeConstraintsForNode(nodeId): Constraint[]
  clear(): void
  toJSON(): ConstraintManagerJSON
  restoreFromJSON(json): void
}

Evaluation Pipeline

1. Keyframe evaluation (per node) 2. Clip instance layering 3. IK solving (per chain) → bone rotation overrides stored 4. Constraint evaluation (per node, sorted by priority) → world transform modifications 5. Final transform applied to Pixi container

Renderer Integration

// In Renderer.#handleTimeChanged():
transformSource.updateIKOverrides(slideId, time)   // solve all chains
sceneRenderer.handleTimeChanged()                    // re-evaluate all nodes
sceneRenderer.applyIKOverrides(rotations)            // apply to containers
sceneRenderer.applyConstraintOverrides()             // apply constraints

// In EvaluatedWorldTransformSource.transformOf(nodeId):
1. composeChain() with IK rotation overrides
2. applyConstraints() if any exist
