Command System API

Quick lookup for command types, dispatcher, and undo stack.

Command Interface

interface Command<Inverse> {
  readonly type: string
  readonly parameters: Readonly<Record<string, unknown>>
  validate(engine: Engine): void
  execute(engine: Engine): Inverse
  toJSON(): Record<string, unknown>
}

type CommandResult<Inverse> =
  | { ok: true;  inverse: Inverse }
  | { ok: false; error: Error }

Dispatcher

class CommandDispatcher {
  dispatch<Inverse>(command: Command<Inverse>): CommandResult<Inverse>
  setOnCommandSucceeded(listener: () => void): void
}

createCommandSystem(logger?): CommandSystem

interface CommandSystem {
  engine: EnginePublic
  dispatcher: CommandDispatcher
  undoStack: UndoStack
  assetLibrarySync: AssetLibrarySync
  materialLibrarySync: MaterialLibrarySync
  shaderLibrarySync: ShaderLibrarySync
}

Undo Stack

interface UndoStackEntry<Inverse> {
  id: string           // unique per entry
  type: string         // command type name
  parameters: Record   // command parameters
  inverse: Inverse     // undo payload
}

class UndoStack {
  entries: readonly UndoStackEntry[]
  record(entry): void           // push newest-first
  clear(): void                 // on project load
  subscribe(listener): () => void  // returns unsubscribe
}

Writing a New Command

  1. Create myCommand.ts in engine/commands/
  2. Define MyCommandParameters, MyCommandInverse interfaces
  3. Implement Command<MyCommandInverse>
  4. In validate(): throw if inputs are invalid (node missing, out of bounds, etc.)
  5. In execute(): call the engine method, return the previous state as inverse
  6. Register in engine/commands/index.ts
Rule: The inverse must capture exactly enough state to undo the command. Not more, not less. Store previous values, not deltas.
