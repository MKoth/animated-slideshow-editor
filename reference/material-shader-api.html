Material & Shader API

Quick lookup for materials, shaders, resolution, and rendering.

MaterialDefinition

class MaterialDefinition {
  readonly id: string
  readonly name: string
  readonly parameters: readonly MaterialParameterDefault[]
  readonly shaderId: string | null
}

interface MaterialParameterDefault {
  key: string
  kind: string       // 'color' | 'number' | 'int' | 'bool' | 'vec2' | 'sampler2D'
  default: MaterialOverrideValue
}

MaterialInstance

interface MaterialInstance {
  materialDefinitionId: string
  overrides: MaterialOverrides
}

type MaterialOverrides = Readonly<Record<string, MaterialOverrideValue>>
type MaterialOverrideValue = string | number | boolean | readonly number[]

defaultMaterial() → { materialDefinitionId: DEFAULT_ID, overrides: {} }

Resolution

// Effective material (tint + opacity)
resolveMaterial(parameters, overrides) → EffectiveMaterialScratch
  tint = overrides.tint ?? definition.default
  opacityMultiplier = clamp(overrides.opacity ?? definition.default, 0, 1)

// Shader uniforms (all non-built-in parameters)
resolveShaderUniforms(parameters, overrides, target, uTimeValue?) → EffectiveShaderScratch
  For each parameter (except built-ins + samplers):
    value = overrides[key] ?? parameter.default
  For each sampler2D:
    assetDefinitionId = overrides[key] ?? null

// Single parameter resolution
resolveParameterValue(parameters, overrides, key) → value | undefined
  override wins over definition default

ShaderDefinition

class ShaderDefinition {
  readonly id: string

EffectiveShaderScratch

interface EffectiveShaderScratch {
  source: string | null          // GLSL source code
  keys: string[]                 // uniform names
  kinds: string[]                // uniform types
  values: MaterialOverrideValue[] // uniform values
  samplers: ShaderSamplerBinding[] // sampler2D bindings
  uTimeValue: number             // current playhead time
}

interface ShaderSamplerBinding {
  key: string                    // uniform name
  assetDefinitionId: string | null  // asset to sample
}
